\section{Mô hình hóa bài toán tìm kiếm (Problem Formulation)}

\subsection{Không gian trạng thái (State Space)}
Trong trò chơi Bloxorz, không gian trạng thái $S$ bao gồm:
\begin{enumerate}
    \item \textbf{Định hướng và vị trí khối hộp}:
    Khối có kích thước $1 \times 1 \times 2$, tại mỗi bước tồn tại ở một trong ba trạng thái định hướng $O$:
    \begin{itemize}
        \item \textbf{Đứng (Standing - S)}: Khối chiếm 1 ô duy nhất tại tọa độ $(r, c)$.
        \item \textbf{Nằm ngang (Lying Horizontally - H)}: Khối chiếm 2 ô liền kề $(r, c)$ và $(r, c + 1)$.
        \item \textbf{Nằm dọc (Lying Vertically - V)}: Khối chiếm 2 ô liền kề $(r, c)$ và $(r + 1, c)$.
    \end{itemize}
    Trường hợp phân tách khối (Split state): Trạng thái lưu tọa độ của cả hai khối lập phương $1 \times 1 \times 1$: $(r_1, c_1)$ và $(r_2, c_2)$ kèm con trỏ chỉ định khối đang được điều khiển.
    
    \item \textbf{Trạng thái cầu (Bridges Configuration)}:
    Giả sử màn chơi có $k$ cây cầu, trạng thái của các cây cầu được biểu diễn dưới dạng tuple hoặc bitmask nhị phân $B = (b_1, b_2, \dots, b_k)$ với $b_i \in \{0, 1\}$ (0: Đóng / Void, 1: Mở / Sàn).
\end{enumerate}
Do đó, một trạng thái tổng quát được biểu diễn chính xác là một bộ:
\begin{equation}
    s = (O, \text{BlockPosition}, B)
\end{equation}

\subsection{Không gian hành động (Actions)}
Tại mỗi bước, tập hành động hợp lệ $A = \{\text{Up}, \text{Down}, \text{Left}, \text{Right}\}$. Khi khối bị tách thành 2 khối nhỏ, hành động mở rộng thêm thao tác hoán đổi quyền điều khiển (\texttt{SwitchControl}).

\subsection{Mô hình chuyển trạng thái (Transition Model)}
Bảng \ref{tab:transition_rules} mô tả quy luật biến đổi tất định vị trí và định hướng khi thực hiện hành động lăn từ tọa độ tham chiếu $(r, c)$:

\begin{table}[htbp]
\centering
\caption{Quy luật chuyển đổi trạng thái khi lăn khối 1×1×2}
\label{tab:transition_rules}
\small
\begin{tabular}{lcccc}
\toprule
\textbf{Trạng thái hiện tại} & \textbf{Up} & \textbf{Down} & \textbf{Left} & \textbf{Right} \\

\subsection{Kiểm tra tính hợp lệ \& Trạng thái kết thúc}
\begin{itemize}
    \item \textbf{Nước đi không hợp lệ (Illegal Move)}: Một hành động lăn bị coi là không hợp lệ nếu bất kỳ phần nào của khối chiếm ô ngoài bản đồ, ô trống (Void), ô cầu đang ở trạng thái Đóng ($b_i = 0$), hoặc khối rơi vào trạng thái Đứng trên ô màu cam dễ vỡ (Fragile Tile).
    \item \textbf{Kiểm tra đích (Goal Test)}: Trạng thái đích đạt được khi và chỉ khi khối ở trạng thái \textbf{Đứng (Standing)} tại tọa độ ô mục tiêu (Goal Hole).
\end{itemize}

\subsection{Quản lý trạng thái lặp (Cycle Detection \& Visited Set)}
Để đảm bảo thuật toán không bị rơi vào chu trình vô hạn, cấu trúc dữ liệu \texttt{visited} được cài đặt bằng bảng băm (\texttt{set} trong Python) với khóa băm (hashable key) là tuple bất biến:
\begin{equation}
    \text{key} = (O, \text{tuple}(\text{pos}), \text{tuple}(B))
\end{equation}

\section{Cài đặt các thuật toán tìm kiếm (Search Algorithms)}

\subsection{Breadth-First Search (BFS)}
\begin{itemize}
    \item \textbf{Nguyên lý}: Sử dụng hàng đợi FIFO (\texttt{collections.deque}) để mở rộng các nút theo từng độ sâu tăng dần.
    \item \textbf{Tính chất}: Với trọng số các bước đi bằng nhau, BFS đảm bảo tính tối ưu (tìm được lời giải có số bước đi ít nhất) và tính đầy đủ (luôn tìm ra lời giải nếu tồn tại).
\end{itemize}

\subsection{Depth-First Search (DFS) / Iterative Deepening Search (IDS)}
\begin{itemize}
    \item \textbf{Nguyên lý}: Sử dụng ngăn xếp LIFO hoặc đệ quy để đi sâu vào các nhánh.
    \item \textbf{Tối ưu hóa IDS}: Do DFS chuẩn có thể bị kẹt ở nhánh sâu vô tận hoặc tìm ra đường đi rất dài, nhóm đã cài đặt \textbf{IDS (Iterative Deepening Search)} bằng cách kết hợp ưu điểm tiết kiệm bộ nhớ của DFS với tính tối ưu độ sâu của BFS.
\end{itemize}

\subsection{Uniform-Cost Search (UCS)}
\begin{itemize}
    \item \textbf{Nguyên lý}: Sử dụng hàng đợi ưu tiên Min-Heap (\texttt{heapq}) mở rộng nút có chi phí đường đi tích lũy $g(n)$ nhỏ nhất trước.
    \item \textbf{Thiết kế hàm chi phí (Cost Function)}: Để tránh việc UCS suy biến thành BFS khi gán mọi bước đi chi phí bằng 1, nhóm đã thiết kế hàm chi phí phi đồng nhất có ý nghĩa thực tế:
    \begin{equation}
        c(s, a, s') = 
        \begin{cases}
            1.0 & \text{Bước lăn trên sàn bình thường} \\
            3.0 & \text{Bước lăn nằm qua ô dễ vỡ (Fragile tile) - do mức độ rủi ro cao} \\
            1.5 & \text{Bước lăn đè lên công tắc (Switch) - chi phí cơ chế kích hoạt} \\
            2.0 & \text{Bước lăn khi đang ở trạng thái phân tách 2 khối nhỏ}
        \end{cases}
    \end{equation}
    \item \textbf{Ý nghĩa trực giác}: Hàm chi phí ưu tiên khối đi qua các con đường an toàn, tránh mạo hiểm qua các ô dễ vỡ nếu có lộ trình thay thế ít rủi ro hơn.
\end{itemize}

\subsection{A* Search}
\begin{itemize}
    \item \textbf{Nguyên lý}: Đánh giá nút theo hàm $f(n) = g(n) + h(n)$, trong đó $g(n)$ là chi phí thực tế từ nút gốc đến $n$, và $h(n)$ là hàm Heuristic ước lượng chi phí còn lại từ $n$ đến đích.
    \item \textbf{Thiết kế hàm Heuristic $h(n)$}:
    \begin{enumerate}
        \item \textbf{Heuristic 1 (Manhattan chia 2)}:
        \begin{equation}
            h_1(n) = \frac{|r_{\text{block}} - r_{\text{goal}}| + |c_{\text{block}} - c_{\text{goal}}|}{2}
        \end{equation}
        Do mỗi bước lăn khối có thể tiến tối đa 2 ô trên lưới, khoảng cách Manhattan chia 2 luôn là một cận dưới hợp lệ (Lower bound). Do đó $h_1(n)$ luôn \textbf{Admissible} ($0 \le h_1(n) \le h^*(n)$).
        
        \item \textbf{Heuristic 2 (Orientation-Aware Heuristic)}:
        Nếu khối đang ở trạng thái Nằm ($H$ hoặc $V$), chắc chắn cần thêm ít nhất 1 bước lăn nữa để dựng đứng vào ô mục tiêu. Vì vậy:
        \begin{equation}
            h_2(n) = h_1(n) + \begin{cases} 0 & \text{nếu } O = S \\ 1 & \text{nếu } O \in \{H, V\} \end{cases}
        \end{equation}
        $h_2(n)$ bám sát chi phí thực tế hơn $h_1(n)$ và vẫn đảm bảo tính chấp nhận được (Admissible) và nhất quán (Consistent).
    \end{enumerate}
\end{itemize}

\subsection{Phân tích \& Nhận xét chuyên sâu}
\begin{enumerate}
    \item \textbf{Về số lượng nút mở rộng và thời gian tìm kiếm}:
    \begin{itemize}
        \item Thuật toán \textbf{A*} thể hiện sự vượt trội rõ rệt về số nút duyệt nhờ hàm heuristic định hướng tìm kiếm trực tiếp về phía ô đích. Số nút duyệt của A* ít hơn từ 40\% đến 70\% so với BFS và UCS trên các bản đồ lớn.
        \item \textbf{BFS và UCS} duyệt theo đường mức (contour) nên phải mở rộng một lượng lớn trạng thái ở các hướng ngược lại với đích.
        \item \textbf{DFS} duyệt rất sâu theo một nhánh ngẫu nhiên, dẫn đến số nút mở rộng biến động mạnh và thời gian tìm kiếm không ổn định.
    \end{itemize}

    \item \textbf{Về tính tối ưu của lời giải}:
    \begin{itemize}
        \item \textbf{BFS} và \textbf{A* (với Heuristic admissible)} luôn đảm bảo tìm được số bước đi ngắn nhất.
        \item \textbf{UCS} tìm được lời giải có tổng chi phí $g(n)$ tối thiểu theo hàm chi phí định sẵn, thành công trong việc tránh các ô có rủi ro cao.
        \item \textbf{DFS} thường trả về các đường đi vòng vèo, số bước đi lớn hơn đáng kể so với nghiệm tối ưu.
    \end{itemize}

    \item \textbf{Về sử dụng bộ nhớ}:
    \begin{itemize}
        \item DFS tiêu tốn ít bộ nhớ nhất trong lý thuyết; tuy nhiên khi áp dụng visited set để tránh lặp chu trình, mức tiêu hao bộ nhớ của DFS tăng lên tương đương các thuật toán khác.
        \item BFS và UCS tiêu tốn nhiều bộ nhớ nhất do Frontier phải lưu trữ toàn bộ các nút ở mức sâu hiện tại.
    \end{itemize}
\end{enumerate}
